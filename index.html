-- 07-reposiciones-hechas.sql -- Historial de remitos de reposición (lo que se bajó del depósito a la camioneta). -- -- La app YA tiene todo el código para subir y bajar estos remitos, pero la -- tabla no existe en la base, así que cada INSERT falla y el remito queda -- sólo en el teléfono. Al refrescar o entrar desde otro dispositivo, no está. -- -- Ejecutar UNA VEZ en el SQL Editor de Supabase (proyecto fanxjjhocudfeiqinbac). -- -- IMPORTANTE sobre los tipos: la app genera los ids con uid(), que devuelve -- texto corto tipo "m0abc12def345" — NO es un UUID. Por eso id, negocio_id y -- origen son TEXT. Si se crean como UUID, el INSERT falla por tipo. CREATE TABLE IF NOT EXISTS reposiciones_hechas ( id TEXT PRIMARY KEY, remito TEXT NOT NULL, fecha DATE NOT NULL, hora TEXT DEFAULT '', negocio_id TEXT, negocio_nombre TEXT DEFAULT '', origen TEXT, origen_nombre TEXT DEFAULT '', usuario TEXT DEFAULT '', items JSONB DEFAULT '[]'::jsonb, unidades INTEGER DEFAULT 0, pendiente_despues INTEGER DEFAULT 0, creado_en TIMESTAMPTZ DEFAULT now() ); -- La app lee con .order("fecha", ascending: false).limit(500) CREATE INDEX IF NOT EXISTS reposiciones_hechas_fecha_idx ON reposiciones_hechas (fecha DESC); -- Para filtrar por local cuando el historial crezca CREATE INDEX IF NOT EXISTS reposiciones_hechas_negocio_idx ON reposiciones_hechas (negocio_id); -- ── Permisos ──────────────────────────────────────────────── -- Mismo criterio que el resto de las tablas de la app: RLS activo con una -- política permisiva, porque el control de quién hace qué lo maneja la app -- por rol, no la base. ALTER TABLE reposiciones_hechas ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "acceso_total_reposiciones_hechas" ON reposiciones_hechas; CREATE POLICY "acceso_total_reposiciones_hechas" ON reposiciones_hechas FOR ALL USING (true) WITH CHECK (true); -- ── Realtime ──────────────────────────────────────────────── -- La app se suscribe a los cambios de esta tabla (postgres_changes) para que -- un remito hecho en un teléfono aparezca en el otro sin recargar. -- Si ya está en la publicación, este bloque no hace nada. DO $$ BEGIN ALTER PUBLICATION supabase_realtime ADD TABLE reposiciones_hechas; EXCEPTION WHEN duplicate_object THEN NULL; WHEN undefined_object THEN NULL; END $$; COMMENT ON TABLE reposiciones_hechas IS 'Un remito por reposición confirmada: qué se bajó del depósito, quién, cuándo y cuánto quedó pendiente.';
